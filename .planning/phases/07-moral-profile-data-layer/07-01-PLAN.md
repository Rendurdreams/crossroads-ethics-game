**Tap-to-rank handlers** (per D-04, D-05):
```javascript
function handleValueTap(value) {
  if (rankedValues.includes(value)) return
  if (rankedValues.length >= 5) return
  setRankedValues(prev => [...prev, value])
}
function handleValueUndo(value) {
  const idx = rankedValues.indexOf(value)
  if (idx === -1) return
  setRankedValues(prev => prev.slice(0, idx))
}
```
Undo clears from-that-rank-onward (Pitfall 5 — always valid prefix).

**Stance handler:**
```javascript
function handleStance(key, value) {

}
```
Once selected, a stance answer can be changed by tapping a different option for the same question (re-sets the key).

**Submit handler** (per D-02, Pitfall 1):
```javascript
async function handleSubmit() {
  if (rankedValues.length < 5 || Object.keys(stances).length < 2) return
  setSubmitting(true)
  setSubmitError(false)
  const { error } = await supabase
    .from('players')
    .update({ moral_values: rankedValues, moral_stances: stances })
    .eq('id', playerId)
  if (!error) {
    localStorage.removeItem('baseline_ranked')
    localStorage.removeItem('baseline_stances')
    navigate(`/play/${sessionId}`)
  } else {
    setSubmitError(true)
    setSubmitting(false)
  }
}
```
MUST await the Supabase update before navigating. Per D-11, D-12.
